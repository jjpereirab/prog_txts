==
resize image

# 2. Resize to exact 800x600 (May distort/stretch image)
ffmpeg -i input.jpg -vf scale=800:600 output.jpg

# 3. Resize width to 800px, auto-calculate height to keep aspect ratio
ffmpeg -i input.jpg -vf scale=800:-1 output.jpg

# 4. Resize height to 600px, auto-calculate width to keep aspect ratio
ffmpeg -i input.jpg -vf scale=-1:600 output.jpg

# 5. Resize to 50% of original size (using input width/height variables)
ffmpeg -i input.jpg -vf "scale=iw/2:ih/2" output.jpg
ffmpeg -i input.jpg -vf "scale=iw*2:ih*2" output.jpg

# 6. Batch Loop: Resize all .jpg files in folder to 1024px width (saves to 'resized' folder)
mkdir -p resized; for i in *.jpg; do ffmpeg -i "$i" -vf scale=1024:-1 "resized/${i%.*}.jpg"; done


== enhance low quality video

== DVD ISO
ver resolucion de un video
	ffprobe -v error -select_streams v:0 -show_entries stream=width,height -of csv=s=x:p=0 input.mp4

reducir resolucion a 900p
	ffmpeg -i input.mp4 -vf "scale=1600:900" -c:a copy output_1600x900.mp4

reducir resolucion a 720p con ancho automatico manteniendo aspect ratio
	ffmpeg -i input.mp4 -vf "scale=-1:720" -c:a copy output_720p.mp4
	
--
convertir mp4 a mpg compatible con DVD
	ffmpeg -i input.mp4 \
	  -target ntsc-dvd -aspect 16:9 \
	  -ac 2 -ar 48000 -b:a 192k \
	  -b:v 6000k -maxrate 8000k -bufsize 1835k \
	  -flags +ilme+ildct \
	  dvd.mpg
	
crear estructura de DVD. Primero crear dvd.xml con
	<dvdauthor dest="dvd" format="ntsc">
	  <vmgm />
	  <titleset>
	    <titles>
	      <pgc>
		<vob file="dvd.mpg"/>
	      </pgc>
	    </titles>
	  </titleset>
	</dvdauthor>

luego
	dvdauthor -x dvd.xml
	dvdauthor -o dvd -T
	
finalmente generar ISO
	genisoimage -dvd-video -o dvd.iso dvd/
