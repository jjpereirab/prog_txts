sudo apt install postgresql

comandos basicos 
	sudo systemctl start postgresql
	sudo systemctl stop postgresql
	sudo systemctl restart postgresql
	sudo systemctl status postgresql

conectarse como usuario default
	sudo -u postgres psql

dentro
	CREATE DATABASE mi_base_de_datos;


	\du # listar usuarios
	\l # listar bases de datos
	
crear usuario, nase de datos, y privilegios
	CREATE USER nuevo_usuario;
	CREATE DATABASE nueva_db OWNER nuevo_usuario;
	GRANT ALL PRIVILEGES ON DATABASE nueva_db TO nuevo_usuario;

cambiar contraseña (el usuario default es 'postgres')
	ALTER USER mi_usuario WITH PASSWORD 'nueva_contraseña';

para entrar a una base de datos mi_db con el usuario predeterminado (<usuario>)
	psql -h localhost mi_db
	psql -U USUARIO -h localhost -d DATABASE
	
	
------------------- eliminar
	DROP DATABASE mi_base_de_datos;
	DROP DATABASE mi_base_de_datos WITH (FORCE);
